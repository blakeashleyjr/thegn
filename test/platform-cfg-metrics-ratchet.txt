# Files in crates/thegn-metrics/src that carry a platform #[cfg].
#
# thegn-metrics is a per-OS leaf crate BY DESIGN (sysfs battery/GPU on Linux,
# sysinfo elsewhere): these files are the platform tables, not debt. A new
# per-OS file is added here with a line saying which OS it serves.
#
# Enforced by `platform_cfgs_are_pinned` in
# crates/thegn-metrics/src/platform_ratchet_tests.rs.

battery.rs
# disk_aggregate.rs — Linux /proc/diskstats + mountinfo reader; sysinfo dedupe elsewhere.
disk_aggregate.rs
# gpu.rs — Linux sysfs (amdgpu/i915) + nvidia-smi vs macOS ioreg accelerator stats.
gpu.rs
# gpu_exec.rs — unix-only fake-helper tests and Linux /proc test helpers.
gpu_exec.rs
# gpu_monitor.rs — unix-only fake-helper lifecycle tests.
gpu_monitor.rs
lib.rs
# net_kind.rs — Linux sysfs link classification; aggregate() is cfg-free.
net_kind.rs
sample.rs
# thermal.rs — macOS Apple-silicon HID temperature sensors (IOHIDEventSystemClient,
# dlsym'd) vs sysinfo Components, which is correct on Linux and Intel Macs.
thermal.rs
# cpu_freq.rs — Linux held sysfs CPU frequency reader; sysinfo frequency fallback elsewhere.
cpu_freq.rs
# vmstat.rs — Linux /proc/vmstat held-fd pread reader; the reader is absent elsewhere.
vmstat.rs
# collect/mod.rs — the only cfg in the collect/ tree: per-OS mod declarations (linux/ today).
collect/mod.rs
# procs.rs — Linux native procfs backend selection; sysinfo elsewhere.
procs.rs
